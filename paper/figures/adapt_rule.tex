  box/.style={draw,rounded corners=2pt,align=center,minimum height=8mm,inner sep=4pt,fill=black!4},

\draw[arr] (gain)--node[above]{yes} (accept); \draw[arr] (gain)--node[right]{no} (stale);
\draw[arr,dashed] (stale)--(sweep);
